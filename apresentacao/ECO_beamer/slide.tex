    % SLIDE 1: Contexto e Problema
    \begin{frame}{Contexto e Problema}
        \textbf{Crescimento do E-commerce:}
        \begin{itemize}
            \item Vendas globais: US\$ 6,56 trilhões (2025) $\rightarrow$ US\$ 8,09 trilhões (2028)
            \item Desafios técnicos: performance, escalabilidade, experiência do usuário
        \end{itemize}

        \vspace{0.4cm}

        \textbf{O Problema:}
        \begin{itemize}
            \item Informação técnica dispersa na literatura
            \item Falta visão integrada: arquitetura + UX + IA
            \item Difícil conexão entre decisões técnicas e impacto financeiro
        \end{itemize}

        \vspace{0.4cm}

        \textbf{Exemplo da Literatura:}
        \begin{itemize}
            \item Alibaba.com: implementação de PWA $\rightarrow$ \textbf{+76\% conversões}
            \item Decisões técnicas podem ter impacto direto na receita
        \end{itemize}
    \end{frame}

    % SLIDE 2: Motivação e Objetivos
    \begin{frame}{Motivação e Objetivos}
        \textbf{Motivação:}
        \begin{itemize}
            \item Impacto econômico de decisões técnicas pouco explorado
            \item Complexidade das escolhas: arquitetura, PWA, IA
            \item Lacuna entre pesquisa acadêmica e prática empresarial
        \end{itemize}

        \vspace{0.3cm}

        \textbf{Objetivo Geral:}
        \begin{itemize}
            \item Analisar estratégias técnicas em e-commerce e discutir impacto na receita
        \end{itemize}

        \vspace{0.3cm}

        \textbf{Objetivos Específicos:}
        \begin{enumerate}
            \item Mapear estratégias arquiteturais e trade-offs
            \item Analisar estratégias de UX/UI para conversão
            \item Avaliar papel da IA e personalização
            \item \textbf{Discutir relações entre decisões técnicas e receita}
            \item Identificar lacunas e oportunidades de pesquisa
        \end{enumerate}
    \end{frame}

    % SLIDE 4: Arquitetura - Detalhes
    \begin{frame}{Revisão: Arquitetura de Software}
        \textbf{Ubur (2023) - Event-Driven vs Monolítico:}
        \begin{itemize}
            \item Monolítico: melhor em baixa carga (menos overhead)
            \item Microsserviços: escalam melhor em alta demanda
            \item \textbf{Conclusão:} contexto determina a melhor escolha
        \end{itemize}

        \vspace{0.4cm}

        \textbf{Zhao et al. (2024) - Mapeamento Sistemático:}
        \begin{itemize}
            \item Análise de 109 estudos sobre performance arquitetural
            \item Falta de padronização em benchmarks
            \item Difícil comparação entre abordagens
        \end{itemize}

        \vspace{0.4cm}

        \textbf{Implicação prática:}
        \begin{itemize}
            \item Não existe ``bala de prata'' arquitetural
            \item Decisão deve considerar: equipe, escala esperada, recursos
        \end{itemize}
    \end{frame}

    % SLIDE 5: UX/UI - Detalhes
    \begin{frame}{Revisão: UX/UI e Performance}
        \textbf{Nawir (2024) - Impacto de Usabilidade:}
        \begin{itemize}
            \item \textbf{70\% da variância} nas taxas de conversão explicada por fatores de UX
            \item Design responsivo e navegação intuitiva são críticos
        \end{itemize}

        \vspace{0.3cm}

        \textbf{Bansal (2024) - SEO e Performance:}
        \begin{itemize}
            \item \textbf{-7\% conversões} para cada segundo adicional de carregamento
            \item Core Web Vitals afetam ranqueamento no Google
        \end{itemize}

        \vspace{0.3cm}

        \textbf{Caso Alibaba.com (PWA):}
        \begin{itemize}
            \item Implementação de Progressive Web App
            \item Resultado: \textbf{+76\% nas conversões}
            \item Melhoria significativa em experiência mobile
        \end{itemize}
    \end{frame}

    % SLIDE 6: IA - Detalhes
    \begin{frame}{Revisão: Inteligência Artificial}
        \textbf{Stamkou et al. (2025) - IA Generativa:}
        \begin{itemize}
            \item ChatGPT para atendimento e descrições de produto
            \item DALL-E para geração de imagens
            \item Potencial de personalização em massa
        \end{itemize}

        \vspace{0.3cm}

        \textbf{Nguyen et al. (2024) - Sistema de Recomendação H\&M:}
        \begin{itemize}
            \item Comparação: LightGBM vs Deep Neural Networks
            \item LightGBM: MAP@50 = 0,06 | MAR@50 = 0,03
            \item DNN: MAP@50 = 0,02 | MAR@50 = 0,01
            \item \textbf{Modelos simples podem superar complexos}
        \end{itemize}

        \vspace{0.3cm}

        \textbf{Achado-chave:} IA tem potencial, mas dados quantitativos sobre impacto em receita são escassos
    \end{frame}

    % SLIDE 7: Síntese da Revisão
    \begin{frame}{Síntese da Revisão Bibliográfica}
        \textbf{Principais achados:}

        \vspace{0.3cm}

        \begin{itemize}
            \item Arquitetura: trade-offs claros, sem solução universal
            \item UX/UI: evidências fortes de impacto em conversão
            \item IA: promessa alta, evidências quantitativas limitadas
            \item Performance: relação direta com métricas de negócio
        \end{itemize}

        \vspace{0.4cm}

        \textbf{Lacunas identificadas:}
        \begin{itemize}
            \item Faltam frameworks integrados de análise
            \item Poucos estudos com dados financeiros reais
            \item Ausência de benchmarks padronizados
            \item Relação quantitativa decisão técnica $\leftrightarrow$ receita pouco explorada
        \end{itemize}
    \end{frame}

    % SLIDE 8: Aplicação Prática - Achievece
    \begin{frame}{Aplicação Prática: Achievece}
        \textbf{Plataforma Achievece (achievece.com):}
        \begin{itemize}
            \item E-commerce de cursos online (educação médica, EUA)
            \item 30 mil visitantes/mês, +700 cursos
            \item Receita: US\$ 2.500/dia
            \item Modelo híbrido: pagamento único + assinaturas
        \end{itemize}

        \vspace{0.4cm}

        \textbf{Por que essa plataforma?}
        \begin{itemize}
            \item Acesso completo a dados e infraestrutura
            \item Volume de tráfego permite análises significativas
            \item Modelo de negócio permite observar diferentes estratégias
        \end{itemize}

        \vspace{0.3cm}

        \textbf{Período de análise:} aproximadamente 1 ano de implementações
    \end{frame}

    % SLIDE 9: Estratégias Implementadas
    \begin{frame}{Estratégias Implementadas}
        \textbf{1. Arquitetura:}
        \begin{itemize}
            \item Migração: WordPress $\rightarrow$ Next.js (serverless na Vercel)
        \end{itemize}

        \vspace{0.2cm}

        \textbf{2. UX/UI:}
        \begin{itemize}
            \item Reorganização da interface (destaque para assinaturas)
            \item Otimização de Core Web Vitals
        \end{itemize}

        \vspace{0.2cm}

        \textbf{3. SEO:}
        \begin{itemize}
            \item Dados estruturados JSON-LD (Organização, Produto, FAQ, Artigo)
            \item +100 páginas personalizadas por estado/profissão
        \end{itemize}

        \vspace{0.2cm}

        \textbf{4. Retenção:}
        \begin{itemize}
            \item Ofertas de retenção no fluxo de cancelamento
            \item Estratégia de upsell pós-compra
        \end{itemize}

        \vspace{0.2cm}

        \textbf{Ferramentas:} Google Analytics 4, Vercel Speed Insights, Search Console
    \end{frame}

    % SLIDE 17: Limitações
    \begin{frame}{Limitações do Trabalho}
        \textbf{Limitações Metodológicas:}
        \begin{itemize}
            \item \textbf{Sem testes A/B}: questões operacionais impediram divisão de tráfego
            \item \textbf{Implementação simultânea}: múltiplas estratégias no mesmo período
            \item \textbf{Amostra única}: apenas uma plataforma analisada
            \item \textbf{Dados históricos perdidos}: sem comparação direta com WordPress
        \end{itemize}

        \vspace{0.4cm}

        \textbf{Implicações:}
        \begin{itemize}
            \item Resultados são \textbf{correlações}, não causalidade comprovada
            \item Não é possível isolar contribuição de cada estratégia
            \item Generalização limitada a contextos similares
        \end{itemize}

        \vspace{0.3cm}

        \textit{O trabalho reconhece essas limitações e apresenta os resultados como observações, não como evidências definitivas.}
    \end{frame}

    % SLIDE 18: Trabalhos Futuros
    \begin{frame}{Trabalhos Futuros}
        \textbf{Direções identificadas:}

        \vspace{0.3cm}

        \begin{itemize}
            \item \textbf{Testes A/B controlados}: isolar impacto de estratégias específicas
            \item \textbf{Estudos longitudinais}: capturar efeitos de longo prazo
            \item \textbf{Múltiplos segmentos}: comparar diferentes tipos de e-commerce
            \item \textbf{Benchmarks padronizados}: facilitar comparação entre estudos
        \end{itemize}

        \vspace{0.4cm}

        \textbf{Recomendação para empresas:}
        \begin{itemize}
            \item Implementar infraestrutura de testes desde o início
            \item Alocar recursos específicos para análise de impacto
            \item Separar períodos de implementação de estratégias
            \item Preservar dados históricos para comparação
        \end{itemize}
    \end{frame}
